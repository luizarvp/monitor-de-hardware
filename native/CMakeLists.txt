cmake_minimum_required(VERSION 3.20)
project(monitor_collector C)
add_library(monitor_collector SHARED src/monitor_collector.c)
target_include_directories(monitor_collector PUBLIC include)
target_compile_definitions(monitor_collector PRIVATE _CRT_SECURE_NO_WARNINGS)
if(WIN32)
  target_link_libraries(monitor_collector PRIVATE user32 advapi32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(monitor_collector_tests tests/test_contract.c)
  target_include_directories(monitor_collector_tests PRIVATE include)
  add_test(NAME collector_contract COMMAND monitor_collector_tests)
endif()
